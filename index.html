Excel (.xlsx) Generator

Download complete submission records and itemized answer tallies directly in Excel format.


function doPost(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var data = JSON.parse(e.postData.contents);
    
    // Create Header if sheet is empty
    if (sheet.getLastRow() === 0) {
      sheet.appendRow([
        "Timestamp", "Name", "Email", "Total Score",
        "R1_Q1 (Hover)", "R1_Q2 (Reach)", "R1_Q3 (Pause)", "R1_Q4 (Move)", "R1_Q5 (Retract)",
        "R2_Q1 Status", "R2_Q1 Mistakes",
        "R2_Q2 Status", "R2_Q2 Mistakes",
        "R2_Q3 Status", "R2_Q3 Mistakes",
        "R2_Q4 Status", "R2_Q4 Mistakes"
      ]);
      sheet.getRange(1, 1, 1, 17).setFontWeight("bold").setBackground("#6d28d9").setFontColor("#ffffff");
    }
    
    sheet.appendRow([
      data.timestamp,
      data.name,
      data.email,
      data.score,
      data.r1Answers[0] || "N/A",
      data.r1Answers[1] || "N/A",
      data.r1Answers[2] || "N/A",
      data.r1Answers[3] || "N/A",
      data.r1Answers[4] || "N/A",
      data.r2Answers[0] ? data.r2Answers[0].status : "N/A",
      data.r2Answers[0] ? data.r2Answers[0].mistakes : 0,
      data.r2Answers[1] ? data.r2Answers[1].status : "N/A",
      data.r2Answers[1] ? data.r2Answers[1].mistakes : 0,
      data.r2Answers[2] ? data.r2Answers[2].status : "N/A",
      data.r2Answers[2] ? data.r2Answers[2].mistakes : 0,
      data.r2Answers[3] ? data.r2Answers[3].status : "N/A",
      data.r2Answers[3] ? data.r2Answers[3].mistakes : 0
    ]);
    
    return ContentService.createTextOutput(JSON.stringify({"result": "success"}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({"result": "error", "error": err.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

How to deploy in Google Sheets:

  1. Open Google Sheet → Extensions → Apps Script.
  2. Paste the code above and save.
  3. Click Deploy → New Deployment.
  4. Select Type: Web app.
  5. Execute as: Me | Who has access: Anyone.
  6. Copy Web App URL and paste it into the field above!
